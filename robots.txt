User-agent: *

# Pages publiques. Cloudflare Pages redirige /page.html vers /page :
# les deux formes doivent être autorisées, sinon Google est bloqué.
# La racine « / » doit être explorable : Google y lit les icônes du site
# pour afficher le favicon dans les résultats de recherche.
Allow: /$
Allow: /accueil.html
Allow: /accueil$
Allow: /login.html
Allow: /login$
Allow: /mentions-legales.html
Allow: /mentions-legales$
Allow: /cgu.html
Allow: /cgu$
Allow: /cgv.html
Allow: /cgv$
Allow: /tarifs.html
Allow: /tarifs$
Allow: /confidentialite.html
Allow: /confidentialite$
Allow: /contact.html
Allow: /contact$

# Fichiers techniques que Google doit pouvoir lire
Allow: /sitemap.xml
Allow: /robots.txt
Allow: /google*.html

# Icônes du site : sans ces lignes, Google ne peut pas récupérer le
# favicon affiché à côté du résultat de recherche.
Allow: /favicon.ico
Allow: /favicon.svg
Allow: /apple-touch-icon.png
Allow: /icon-192.png
Allow: /icon-512.png
Allow: /site.webmanifest

# Tout le reste est hors limites
Disallow: /

Sitemap: https://ecg-prepa.fr/sitemap.xml
